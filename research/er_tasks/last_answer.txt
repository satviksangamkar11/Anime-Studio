def editorial_from_lock(lock, shot_id, lang):
    fps = lock.get("fps", 24)
    frames = timeline_lock_io.frames_for(lock, shot_id)
    retime = timeline_lock_io.retime_for(lock, shot_id, lang)
    return frames / fps * retime


def apply_lock(rows, lock, lang):
    out = []
    for row in rows:
        r = dict(row)
        r["editorial_duration_sec"] = editorial_from_lock(lock, r["id"], lang)
        out.append(r)
    return out
